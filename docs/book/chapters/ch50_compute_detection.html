50.1 Compute Modes

# From compute.py
COMPUTE_AUTO = "auto"
COMPUTE_CPU = "cpu"
COMPUTE_CUDA = "cuda"
COMPUTE_DML = "dml"  # DirectML (deprecated)

def detect(force=False) -> dict:
    """Detect available compute back-ends."""
    result = {"cpu": True, "cuda": False, "dml": False}

    # CUDA requires both NVIDIA GPU AND downloaded runtime
    if _has_nvidia_gpu() and _runtime_installed():
        result["cuda"] = True

    return result

def _has_nvidia_gpu() -> bool:
    """Check for NVIDIA GPU via nvidia-smi."""
    try:
        result = subprocess.run(
            ["nvidia-smi", "--query-gpu=name", "--format=csv,noheader"],
            capture_output=True, text=True, timeout=10,
        )
        return result.returncode == 0 and bool(result.stdout.strip())
    except Exception:
        return False

def resolve_compute(mode: str) -> str:
    """Resolve 'auto' to the best available mode."""
    if mode != COMPUTE_AUTO:
        return mode
    det = detect()
    if det["cuda"]:
        return COMPUTE_CUDA
    return COMPUTE_CPU

def available_compute() -> list[str]:
    """Return list of available compute modes."""
    det = detect()
    modes = [COMPUTE_CPU]
    if det["cuda"]:
        modes.append(COMPUTE_CUDA)
    return modes

def provider_for(mode: str) -> str:
    """Map compute mode to sherpa-onnx provider name."""
    return {COMPUTE_CUDA: "cuda", COMPUTE_CPU: "cpu"}.get(mode, "cpu")

50.2 Two-Step GPU Requirement

Key design decision: Even with an NVIDIA GPU, CUDA compute is NOT offered until the user explicitly downloads the GPU runtime from Settings → Compute. This prevents the app from failing on machines with NVIDIA GPUs but without the required CUDA libraries. 
# Test from test_compute.py
def test_cuda_not_offered_without_downloaded_runtime():
    # Even with NVIDIA GPU, no GPU option until runtime is downloaded
    self._patch(runtime_installed=False, nvidia=True)
    det = compute.detect(force=True)
    self.assertFalse(det["cuda"])
    self.assertNotIn("cuda", compute.available_compute())

50.3 Key Takeaways

  • GPU detection uses nvidia-smi subprocess call
  • CUDA requires two conditions: NVIDIA GPU present AND runtime downloaded
  • resolve_compute("auto") picks the best available mode
  • The Recording Dialog shows a CUDA GPU (OmniVoice) option when NVIDIA is detected
  • DML (DirectML) is no longer offered
