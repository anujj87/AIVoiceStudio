26.1 ctypes Basics

import ctypes

# Load a shared library
libc = ctypes.CDLL("libc.so.6")  # Linux
# libc = ctypes.CDLL("msvcrt.dll")  # Windows

# Call C functions
libc.printf(b"Hello from C!\n")

# Type mapping
ctypes.c_int(42)        # C int
ctypes.c_double(3.14)   # C double
ctypes.c_char_p(b"hi")  # C char*
ctypes.c_bool(True)     # C bool

26.2 How sherpa-onnx Works

# sherpa-onnx is a Python package wrapping C++ ONNX inference
# AIVS accesses it like any Python library:
import sherpa_onnx

# Create TTS config
config = sherpa_onnx.OfflineTtsConfig(
    model=sherpa_onnx.OfflineTtsModelConfig(
        vits=sherpa_onnx.OfflineTtsVitsModelConfig(
            model="model.onnx",
            tokens="tokens.txt",
            data_dir="espeak-ng-data",
        ),
    ),
)

# Create engine and synthesize
tts = sherpa_onnx.OfflineTts(config)
audio = tts.generate(text="Hello world", sid=0, speed=1.0)

# audio.samples is a numpy array
import numpy as np
samples = np.array(audio.samples, dtype=np.int16)

26.3 NumPy and C

# NumPy arrays are stored as C arrays in memory
import numpy as np

arr = np.array([1, 2, 3, 4], dtype=np.int16)

# Access the C pointer
ptr = arr.ctypes.data  # Memory address
print(f"Array at {ptr}, {arr.shape} elements of {arr.dtype}")

# This is how audio data flows through the system:
# sherpa-onnx → C++ inference → numpy array → wxPython playback

26.4 Key Takeaways

  • ctypes enables calling C shared libraries from Python
  • NumPy provides the bridge between Python and C-level array operations
  • AIVS uses sherpa-onnx (C++ ONNX inference) and NumPy for audio processing
  • The audio pipeline: Text → sherpa-onnx (C++) → numpy array → WAV file
