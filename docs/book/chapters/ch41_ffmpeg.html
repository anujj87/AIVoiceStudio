41.2 AIVS FFmpeg Management

# From audio/ffmpeg.py
def ensure_ffmpeg() -> str | None:
    """Find or return None if FFmpeg is not installed."""
    ffmpeg_path = os.path.join(paths.user_data_dir(), "ffmpeg", "ffmpeg.exe")
    if os.path.isfile(ffmpeg_path):
        return ffmpeg_path
    return None

def convert_wav(input_path, output_path, format, ffmpeg_exe):
    """Convert WAV to MP3 or FLAC using FFmpeg."""
    import subprocess
    cmd = [ffmpeg_exe, "-i", input_path, "-y", output_path]
    subprocess.run(cmd, capture_output=True, check=True)

# Usage in RecordingDialog
if fmt != "wav":
    ffmpeg_exe = ffmpeg_mod.ensure_ffmpeg()
    if not ffmpeg_exe:
        # Offer to download FFmpeg
        answer = wx.MessageBox(
            "FFmpeg is required for MP3/FLAC. Download now?",
            "FFmpeg required",
            style=wx.YES_NO | wx.ICON_QUESTION,
        )
        if answer == wx.YES:
            ffmpeg_mod.download_ffmpeg()

41.3 Key Takeaways

  • WAV is always available (Python stdlib); MP3/FLAC need FFmpeg
  • FFmpeg is downloaded on-demand to the user's AppData folder
  • AIVS uses subprocess.run() to invoke FFmpeg for conversion
  • The format selection happens in the Recording Dialog
