4.4 Scope — The LEGB Rule

Python looks up variable names in this order: Local → Enclosing → Global → Built-in.

# Global scope
APP_NAME = "AI Voice Studio"

def outer():
    # Enclosing scope
    version = "2.0"

    def inner():
        # Local scope
        print(APP_NAME)    # Global — found
        print(version)     # Enclosing — found
        print(len("hi"))   # Built-in — found
        name = "inner"    # Local variable

    inner()

outer()

global and nonlocal

# global — modify a global variable from inside a function
counter = 0

def increment():
    global counter
    counter += 1

increment()
print(counter)  # 1

# nonlocal — modify an enclosing scope variable
def make_counter():
    count = 0
    def increment():
        nonlocal count
        count += 1
        return count
    return increment

counter = make_counter()
print(counter())  # 1
print(counter())  # 2

4.8 Recursive Functions

# Factorial (classic example)
def factorial(n: int) -> int:
    if n <= 1:
        return 1
    return n * factorial(n - 1)

# Recursive directory traversal (used in AIVS)
import os

def find_all_files(directory: str) -> list[str]:
    """Recursively find all files in a directory."""
    result = []
    for entry in os.listdir(directory):
        full_path = os.path.join(directory, entry)
        if os.path.isfile(full_path):
            result.append(full_path)
        elif os.path.isdir(full_path):
            result.extend(find_all_files(full_path))
    return result

4.9 Key Takeaways

  • Use type hints on parameters and return values for clarity
  • Docstrings should explain what, not how
  • Default parameters must be immutable (None for mutable defaults)
  • *args and **kwargs enable flexible function signatures
  • Lambdas are for short, throwaway functions — use def for anything complex
  • Avoid global and nonlocal when possible — they make code harder to reason about
  • Functions should do one thing and do it well (Single Responsibility Principle)
