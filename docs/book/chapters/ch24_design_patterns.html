24.1 Singleton Pattern

# Ensures only one instance exists
class Settings:
    _instance = None
    _lock = threading.Lock()

    def __new__(cls, *args, **kwargs):
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
        return cls._instance

# From AIVS — the addon manager is a singleton
_default_manager: AddonManager | None = None
_manager_lock = threading.Lock()

def get_addon_manager() -> AddonManager:
    global _default_manager
    if _default_manager is None:
        with _manager_lock:
            if _default_manager is None:
                _default_manager = AddonManager()
    return _default_manager

24.2 Observer Pattern

EVT_DOWNLOAD_PROGRESS = wx.PyEventBinder(_DOWNLOAD_PROGRESS, 1)

class DownloadProgressEvent(wx.PyCommandEvent):
    def __init__(self, name: str, done: int, total: int):
        super().__init__(_DOWNLOAD_PROGRESS)
        self.name = name
        self.done = done
        self.total = total

24.3 Strategy Pattern

# Different document parsers for different file types
class Parser(Protocol):
    def parse(self, path: str) -> Document: ...

class PdfParser:
    def parse(self, path): ...

class DocxParser:
    def parse(self, path): ...

# Strategy selection at runtime
parsers = {
    ".pdf": PdfParser(),
    ".docx": DocxParser(),
}

def parse_document(path: str) -> Document:
    ext = os.path.splitext(path)[1].lower()
    parser = parsers.get(ext)
    if parser is None:
        raise ParseError(f"Unsupported: {ext}")
    return parser.parse(path)

24.4 Factory Pattern

# From tts/engine.py — factory that creates the right engine
def get_engine(voice_entry: dict) -> TtsEngine:
    """Factory: create the appropriate engine for a voice."""
    engine_type = voice_entry.get("engine", "vits")

    if engine_type == "xtts":
        client = CloneClient()
        return XtTsCloneEngine(voice_entry, client)
    elif engine_type == "omnivoice":
        return OmniVoiceEngine(voice_entry)
    elif engine_type == "omnivoice_server":
        return OmniVoiceServerEngine(voice_entry)
    else:
        return SherpaEngine(voice_entry)  # Default: sherpa-onnx

24.5 Key Takeaways

  • Singleton for shared state (Settings, AddonManager)
  • Observer for thread-GUI communication (wx events)
  • Strategy for swappable algorithms (document parsers)
  • Factory for object creation based on type (TTS engines)
  • Python makes patterns simpler — often a dict or function replaces a full class hierarchy
