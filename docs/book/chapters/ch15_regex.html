15.1 Regex Basics

import re

# Simple patterns
text = "Voice af_bella has sample rate 24000"

# re.search — find first match anywhere in string
match = re.search(r'\d+', text)
print(match.group())  # "24000"

# re.findall — find ALL matches as strings
numbers = re.findall(r'\d+', text)
print(numbers)  # ["24000"]

# re.match — match only at the START of string
match = re.match(r'Voice (\w+)', text)
print(match.group(1))  # "af_bella" (captured group)

# re.sub — substitute matches
clean = re.sub(r'\s+', ' ', "too   many    spaces")
print(clean)  # "too many spaces"

15.3 Common Patterns

# Validate file extension
def is_valid_docx(path: str) -> bool:
    return bool(re.match(r'\.docx$', path, re.IGNORECASE))

# Extract version numbers
text = "Version 2.0.1 (build 20260822)"
version = re.search(r'Version (\d+\.\d+\.\d+)', text)
print(version.group(1))  # "2.0.1"

# Split on multiple delimiters
parts = re.split(r'[;,\s]+', "Piper, Kokoro ; Matcha")
print(parts)  # ['Piper', 'Kokoro', 'Matcha']

# Clean up whitespace in text
text = "Too\n\n\nmany\n\nnewlines"
clean = re.sub(r'\n{3,}', '\n\n', text)

# Named groups
pattern = r'(?P<tts>\w+)/(?P<lang>\w+)/(?P<variant>\w+)'
match = re.match(pattern, "piper/en_US/medium")
if match:
    print(match.group("tts"))      # "piper"
    print(match.group("lang"))     # "en_US"
    print(match.group("variant"))  # "medium"

15.4 Regex in AIVS

# From util.py — filename sanitization
import re
def sanitize_filename(name: str, max_length: int = 80) -> str:
    name = re.sub(r'[<>:"/\\|?*]', '_', name)
    name = re.sub(r'\s+', ' ', name).strip()
    if len(name) > max_length:
        name = name[:max_length].rstrip()
    return name

# From documents/daisy_builder.py — DAISY ID sanitization
def _sanitize_id(text: str) -> str:
    s = re.sub(r"[^a-zA-Z0-9_-]", "_", text.lower())
    s = re.sub(r"_+", "_", s).strip("_")
    return s[:60] or "item"

# From documents/splitter.py — sentence splitting
def _split_sentences(text: str) -> list[str]:
    sentences = re.split(r'(?<=[.!?])\s+', text)
    return [s.strip() for s in sentences if s.strip()]

# Compiled patterns (faster for repeated use)
_FILE_EXT = re.compile(r'(\.\w+)$')
def get_extension(path: str) -> str:
    m = _FILE_EXT.search(path)
    return m.group(1) if m else ""

15.5 Key Takeaways

  • re.search() finds anywhere; re.match() finds at start; re.findall() finds all
  • Raw strings (r'...') are essential for regex patterns
  • Compiled patterns (re.compile()) are faster for repeated use
  • AIVS uses regex for filename sanitization, text splitting, and pattern matching
