def split_document(doc: Document, mode: str) -> list[Segment]:
    """Split a parsed document into TTS segments."""
    if mode == MODE_PAGE_WITH_H1:
        return _split_by_h1_then_pages(doc)
    elif mode == MODE_PAGE_ONLY:
        return _split_by_pages(doc)
    elif mode == MODE_CHAPTER_ONLY:
        return _split_by_headings(doc)
    elif mode == MODE_FIXED_SIZE:
        return _split_by_size(doc, chunk_size=2000)
    return []

46.3 The Splitting Algorithms

def _split_by_h1_then_pages(doc: Document) -> list[Segment]:
    """Split at H1 headings, then at page breaks."""
    segments = []
    current_text = []
    current_title = "Introduction"

    for block in doc.blocks:
        if block.level == 1 and current_text:
            segments.append(Segment(
                index=len(segments) + 1,
                title=current_title,
                text="\n\n".join(current_text),
            ))
            current_text = []
        if block.level == 1:
            current_title = block.heading
        current_text.append(block.text)

    if current_text:
        segments.append(Segment(
            index=len(segments) + 1,
            title=current_title,
            text="\n\n".join(current_text),
        ))

    return segments

def _split_by_size(doc: Document, chunk_size: int = 2000) -> list[Segment]:
    """Split text into fixed-size chunks at sentence boundaries."""
    import re
    full_text = "\n\n".join(b.text for b in doc.blocks)
    sentences = re.split(r'(?<=[.!?])\s+', full_text)

    segments = []
    current_chunk = []
    current_len = 0

    for sent in sentences:
        if current_len + len(sent) > chunk_size and current_chunk:
            segments.append(Segment(
                index=len(segments) + 1,
                title=f"Chunk {len(segments) + 1}",
                text=" ".join(current_chunk),
            ))
            current_chunk = []
            current_len = 0
        current_chunk.append(sent)
        current_len += len(sent)

    if current_chunk:
        segments.append(Segment(
            index=len(segments) + 1,
            title=f"Chunk {len(segments) + 1}",
            text=" ".join(current_chunk),
        ))

    return segments

46.4 DAISY Chapter Splitting

def split_daisy_chapters(blocks: list) -> list[Segment]:
    """Split document into chapters for DAISY book output."""
    segments = []
    for block in blocks:
        if block.level <= 2:  # H1 and H2 become chapters
            segments.append(Segment(
                index=len(segments) + 1,
                title=block.heading,
                text=block.text,
            ))
    return segments

46.5 Key Takeaways

  • 4 splitting modes: page+heading, page-only, heading-only, fixed-size
  • Each mode produces a list of Segment objects (index, title, text)
  • DAISY books use chapter-level splitting (H1/H2 headings)
  • Fixed-size splitting respects sentence boundaries
  • The splitter is format-agnostic — it works on parsed Document objects
