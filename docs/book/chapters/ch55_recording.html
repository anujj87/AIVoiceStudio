55.2 Compute-Mode Voice Filtering

    """Show only TTS engines compatible with selected compute."""
    is_omni = self._is_omnivoice_compute()
    self._voices = []
    for v in self._all_voices:
        engine = v.get("engine", "vits")
        if is_omni:
            if engine in _OMNIVOICE_ENGINES:
                self._voices.append(v)
        else:
            if engine in _ONNX_TTS_ENGINES:
                self._voices.append(v)

55.3 Recording Lifecycle

# Start → creates SynthesisWorker thread
def _on_start(self, _):
    voice = self._selected_voice()
    params = self._current_params()
    start_index = project.first_pending_index(self.project_dir)

    self._worker = SynthesisWorker(
        segments=self._segments[start_index:],
        voice_entry=voice, params=params,
        output_dir=self.project_dir,
        start_index=start_index,
        on_segment_done=lambda i, t, p: self._segment_done(i, t, p),
        on_all_done=lambda: wx.PostEvent(self, SynthFinishedEvent()),
        on_error=lambda msg: wx.PostEvent(self, SynthErrorEvent(msg)),
        cancel_event=self._cancel_event,
        pause_event=self._pause_event,
        on_status=lambda msg: wx.PostEvent(self, SynthStatusEvent(msg)),
    )
    self._worker.start()

# Pause → sets pause event
def _on_pause(self, _):
    self._worker.pause()
    self._pause_event.set()

# Stop → sets cancel event
def _on_stop(self, _):
    self._worker.cancel()
    self._cancel_event.set()

# Close → saves TTS settings, ends modal
def _on_close(self, _):
    self._save_tts_to_project()
    self.EndModal(wx.ID_CLOSE)

55.4 Key Takeaways

  • The Recording Dialog is the most complex GUI component in AIVS
  • Compute mode filters which TTS engines are available
  • Segments resume from first_pending_index — previously recorded segments are skipped
  • Pause/Resume/Stop control the background synthesis thread
  • TTS settings are saved to project.json on every start and close
