29.2 The add_labeled Helper (AIVS)

# From gui/a11y.py — reusable helper for label+control pairs
def add_labeled(parent, sizer, label_text, control, **kwargs):
    """Add a label-control pair to a sizer."""
    label = wx.StaticText(parent, label=label_text + ":")
    control.SetName(label_text)  # Accessibility name
    sizer.Add(label, 0, wx.ALIGN_CENTER_VERTICAL | wx.ALL, 4)
    sizer.Add(control, 1, wx.EXPAND | wx.ALL, 4)
    return label

# Usage
add_labeled(self, grid, "TTS engine", self.tts_combo)
add_labeled(self, grid, "Language", self.lang_combo)

29.3 Slider with Value Display

# From recording_dialog.py — slider with live value label
def _slider(self, grid, name: str, lo: float, hi: float) -> wx.Slider:
    label = wx.StaticText(self, label=name + ":")
    slider = wx.Slider(
        self,
        minValue=int(lo * 100),
        maxValue=int(hi * 100),
        value=int(1.0 * 100),
    )
    value_label = wx.StaticText(self, label="1.00")
    slider.value_label = value_label  # Attach label to slider

    # Update label when slider moves
    slider.Bind(
        wx.EVT_SLIDER,
        lambda evt, s=slider, vl=value_label, n=name: (
            vl.SetLabel(f"{s.GetValue() / 100.0:.2f}"),
            s.SetName(f"{n}: {s.GetValue() / 100.0:.2f}"),
        ),
    )

    box = wx.BoxSizer(wx.HORIZONTAL)
    box.Add(slider, 1, wx.EXPAND)
    box.Add(value_label, 0, wx.LEFT | wx.ALIGN_CENTER_VERTICAL, 6)
    grid.Add(label, 0, wx.ALIGN_CENTER_VERTICAL)
    grid.Add(box, 1, wx.EXPAND)
    return slider

29.4 Key Takeaways

  • wxPython provides native controls for every common UI pattern
  • ComboBox with CB_READONLY is the preferred dropdown (stores client data)
  • Slider values are integers — divide by 100 for float parameters
  • AIVS uses the add_labeled helper for consistent label+control pairs
