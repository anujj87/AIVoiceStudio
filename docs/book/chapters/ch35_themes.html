35.1 Theme Manager in AIVS

# From gui/theme.py
_LIGHT = {"bg": "#ffffff", "fg": "#1a1a1a",
          "control_bg": "#ffffff", "control_fg": "#1a1a1a"}
_DARK = {"bg": "#1e1e1e", "fg": "#e8e8e8",
         "control_bg": "#2d2d2d", "control_fg": "#e8e8e8"}

_state = {"theme": THEME_SYSTEM, "colors": _LIGHT}
_lock = threading.Lock()

def apply_theme(top: wx.Window, theme: str) -> None:
    """Apply theme to window and all descendants."""
    resolved = _resolve(theme)
    colors = _DARK if resolved == THEME_DARK else _LIGHT
    with _lock:
        _state["theme"] = theme
        _state["colors"] = colors
    _paint(top, colors)
    for child in _descendants(top):
        _paint(child, colors)
    top.Refresh()

def _resolve(theme: str) -> str:
    """Resolve 'system' to light or dark based on OS."""
    if theme == THEME_SYSTEM:
        try:
            appearance = wx.SystemSettings.GetAppearance()
            return THEME_DARK if appearance.IsDark() else THEME_LIGHT
        except Exception:
            return THEME_LIGHT
    return theme

35.2 Applying Colors to Widgets

def _paint(window: wx.Window, colors: dict) -> None:
    try:
        # Containers get background color
        if isinstance(window, (wx.Panel, wx.Dialog, wx.Frame, wx.ScrolledWindow)):
            window.SetBackgroundColour(colors["bg"])

        # Text labels get foreground color
        elif isinstance(window, (wx.StaticText, wx.CheckBox, wx.RadioButton)):
            window.SetForegroundColour(colors["fg"])

        # Input controls get both
        elif isinstance(window, (wx.TextCtrl, wx.ComboBox, wx.ListBox)):
            window.SetBackgroundColour(colors["control_bg"])
            window.SetForegroundColour(colors["control_fg"])
    except Exception:
        pass  # Some widgets don't support color changes

35.3 Key Takeaways

  • Three themes: Light, Dark, System (auto-detect OS preference)
  • wx.SystemSettings.GetAppearance() detects dark mode on Windows 10/11
  • Colors are applied by walking the widget tree and setting background/foreground
  • Buttons keep native look — only panels, text, and inputs are themed
  • Theme state is stored in a thread-safe dict with locking
