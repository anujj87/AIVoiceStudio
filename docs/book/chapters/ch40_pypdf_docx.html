40.1 PDF Parsing with pypdf

from pypdf import PdfReader

reader = PdfReader("document.pdf")
print(f"Pages: {len(reader.pages)}")

# Extract text from each page
for i, page in enumerate(reader.pages):
    text = page.extract_text()
    print(f"Page {i+1}: {len(text)} chars")

40.2 DOCX Parsing with python-docx

from docx import Document

doc = Document("document.docx")

# Extract paragraphs
for para in doc.paragraphs:
    if para.text.strip():
        style = para.style.name  # "Heading 1", "Normal", etc.
        print(f"[{style}] {para.text[:80]}")

# Extract tables
for table in doc.tables:
    for row in table.rows:
        cells = [cell.text for cell in row.cells]
        print(" | ".join(cells))

40.3 AIVS Document Parser

# From documents/parsers.py
class ParseError(Exception):
    """Raised when a document cannot be parsed."""

def parse_document(path: str) -> Document:
    """Parse a document and return structured content."""
    ext = os.path.splitext(path)[1].lower()
    if ext == ".pdf":
        return _parse_pdf(path)
    elif ext == ".docx":
        return _parse_docx(path)
    elif ext in (".txt", ".md"):
        return _parse_text(path)
    elif ext in (".html", ".htm"):
        return _parse_html(path)
    else:
        raise ParseError(f"Unsupported file type: {ext}")

class Document:
    """Parsed document with blocks of text."""
    def __init__(self):
        self.blocks: list[TextBlock] = []
        self.metadata: dict = {}

class TextBlock:
    """A section of text with a heading and content."""
    def __init__(self, heading: str, text: str, level: int = 0):
        self.heading = heading
        self.text = text
        self.level = level

40.4 Key Takeaways

  • pypdf extracts text from PDF files page by page
  • python-docx extracts paragraphs and tables from DOCX files
  • AIVS supports 7 file formats: PDF, DOCX, TXT, MD, HTML, HTM, EPUB
  • The parser returns a Document with structured TextBlocks
  • ParseError is raised for unsupported or corrupt files
