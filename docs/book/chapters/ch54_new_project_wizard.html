54.2 Finish Flow

def _on_finish(self, _):
    # 1. Get user selections
    name = self.page_details.name_ctrl.GetValue().strip()
    ptype = self.page_details.selected_project_type()
    mode = self.page_mode.selected()

    # 2. Clipboard mode — no document needed
    if ptype == PROJECT_TYPE_CLIPBOARD:
        # Create project with clipboard text
        project.create_project(pdir, name, "<clipboard>", ...)
        self.EndModal(wx.ID_OK)
        return

    # 3. Parse document
    wx.BeginBusyCursor()
    doc = parse_document(self.source_path)
    if ptype in (PROJECT_TYPE_DAISY_AUDIO, PROJECT_TYPE_DAISY_AUDIO_TEXT):
        segments = split_daisy_chapters(doc.blocks)
    else:
        segments = split_document(doc, mode)
    wx.EndBusyCursor()

    # 4. Create project and save
    project.create_project(pdir, name, self.source_path, mode, segments)
    self.settings.add_recent_project(name, pdir)
    self.EndModal(wx.ID_OK)

    # 5. Open recording window
    dlg = RecordingDialog(self.GetParent(), pdir, self.settings, self.store)
    dlg.ShowModal()

54.3 Key Takeaways

  • Two pages: Details (name, type, document, punctuation) and Mode (splitting)
  • 4 project types affect document processing and output format
  • Clipboard mode allows recording without a document
  • The wizard automatically opens the Recording Dialog after project creation
  • Busy cursor is shown during document parsing
