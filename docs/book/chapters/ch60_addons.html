60.2 AddonManager

class AddonManager:
    """Discovers, loads, and manages addons."""

    def discover(self) -> list[AddonManifest]:
        """Scan addons directory and load manifests."""
        for name in os.listdir(self._dir):
            manifest_path = os.path.join(self._dir, name, "manifest.json")
            if os.path.isfile(manifest_path):
                with open(manifest_path) as f:
                    data = json.load(f)
                manifest = AddonManifest(manifest_path, data)
                if manifest.is_compatible:
                    self._addons[name] = manifest

    def load_addon(self, name: str) -> bool:
        """Load addon's Python module and register its engine."""
        manifest = self._addons[name]
        addon_dir = manifest.addon_dir

        # Dynamic import
        spec = importlib.util.spec_from_file_location(
            f"_aivs_addon_{name}",
            os.path.join(addon_dir, "__init__.py"),
        )
        module = importlib.util.module_from_spec(spec)
        sys.modules[f"_aivs_addon_{name}"] = module
        spec.loader.exec_module(module)

        # Register engine
        if manifest.engine_id:
            self._register_engine(manifest, addon_dir)

60.3 Key Takeaways

  • Addons follow the NVDA addon architecture pattern
  • manifest.json declares name, version, engine, dependencies
  • Dynamic imports via importlib load addon code at runtime
  • Addons can provide new TTS engines, voices, and punctuation modes
  • The addon system is thread-safe with RLock protection
