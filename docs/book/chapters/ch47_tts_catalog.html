Chapter 47

TTS Catalog & Model Store

Current implementation (2026.3.1). The JSON catalog now describes nine TTS families — Piper, Kokoro-82M, Kitten TTS, VITS zh-AISHELL3, Matcha-TTS, OmniVoice, OmniVoice Server, SAPI5 and the Windows Core voices. The Voice Lab clone engines (Pocket TTS, Bark, F5-TTS) are deliberately not in this file: their voices come from the installed Python package, so they live in voicelab/engines.py and are emitted in the same catalog shape — see Chapter 76. Engines that are pip packages also carry a requires_package field, which the panels turn into an installed / not-installed state (see Chapter 77).

Learning Objectives

  • Understand the TTS catalog data structure
  • Master the ModelStore for tracking installed voices
  • See how voice file resolution works

47.1 The TTS Catalog

# From tts/catalog.py — a large tuple of TTS engine definitions
# Each entry has: id, name, engine, license, languages → variants → voices

def get_tts_list() -> list[dict]:
    """Return all TTS engines from the catalog."""
    return list(_CATALOG)

def find_tts(tts_id: str) -> dict | None:
    """Find a TTS engine by ID."""
    for tts in _CATALOG:
        if tts["id"] == tts_id:
            return tts
    return None

def find_language(tts: dict, lang_code: str) -> dict | None:
    for lang in tts.get("languages", []):
        if lang["code"] == lang_code:
            return lang
    return None

def find_variant(tts, lang_code, variant_id) -> dict | None:
    lang = find_language(tts, lang_code)
    if lang:
        for v in lang.get("variants", []):
            if v["id"] == variant_id:
                return v
    return None

def find_voice(tts, lang_code, variant_id, voice_id) -> dict | None:
    variant = find_variant(tts, lang_code, variant_id)
    if variant:
        for v in variant.get("voices", []):
            if v["id"] == voice_id:
                return v
    return None

47.2 The ModelStore

# From tts/models.py
class ModelStore:
    """Manages which voice models are installed on disk."""

    def __init__(self, state_file=None):
        self._state_file = state_file or self._default_state_file()
        self._state = self._load_state()

    def variant_dir(self, tts_id, lang, variant) -> str:
        """models/piper/en_US/medium"""
        return os.path.join(paths.models_dir(), tts_id, lang, variant)

    def is_variant_installed(self, tts_id, lang, variant) -> bool:
        return os.path.isdir(self.variant_dir(tts_id, lang, variant))

    def mark_installed(self, tts_id, lang, variant, path):
        self._state.setdefault(tts_id, {})
        self._state[tts_id].setdefault(lang, {})
        self._state[tts_id][lang][variant] = {"path": path}
        self._save_state()

    def installed_voices(self) -> list[dict]:
        """Return flat list of all installed voice entries."""
        voices = []
        for tts_id, langs in self._state.items():
            tts = catalog.find_tts(tts_id)
            for lang_code, variants in langs.items():
                for var_id in variants:
                    variant = catalog.find_variant(tts, lang_code, var_id)
                    if variant:
                        for voice in variant.get("voices", []):
                            entry = dict(voice)
                            entry["tts"] = tts_id
                            entry["language"] = lang_code
                            entry["variant"] = var_id
                            entry["dir"] = self.variant_dir(tts_id, lang_code, var_id)
                            voices.append(entry)
        return voices

47.3 Voice File Resolution

# From tts/models.py — resolve voice entry → actual file paths
def resolve_voice_files(entry: dict) -> dict[str, str]:
    """Map a voice entry dict to actual file paths on disk."""
    vdir = entry.get("dir", "")
    engine = entry.get("engine", "vits")

    result = {
        "model": "", "tokens": "", "data_dir": "",
        "lexicon": "", "vocoder": "", "voices_file": "",
        "engine": engine,
    }

    if not vdir or not os.path.isdir(vdir):
        return result

    # Find model file
    for f in os.listdir(vdir):
        if f.endswith(".onnx"):
            result["model"] = os.path.join(vdir, f)
            break

    # Find tokens file
    tokens = os.path.join(vdir, "tokens.txt")
    if os.path.isfile(tokens):
        result["tokens"] = tokens

    # Find voices file (for multi-speaker models)
    voices = os.path.join(vdir, "voices.bin")
    if os.path.isfile(voices):
        result["voices_file"] = voices

    # Shared espeak-ng-data
    shared = os.path.join(paths.models_dir(), "shared", "espeak-ng-data")
    if os.path.isdir(shared) and entry.get("frontend") != "char":
        result["data_dir"] = shared

    return result

47.4 Key Takeaways

  • The catalog is a nested data structure: TTS → language → variant → voice
  • ModelStore tracks which variants are installed on disk
  • resolve_voice_files() maps voice entries to actual file paths
  • Shared resources (espeak-ng-data) are stored in a common directory
  • The catalog has 14 TTS engines with hundreds of voices across dozens of languages
