37.1 NumPy Arrays — The Audio Building Block

# Audio properties

# Create synthetic audio (sine wave)

freq = 440  # A4 note
wave = np.sin(2 * np.pi * freq * t)
samples_16bit = (wave * 32767).astype(np.int16)

37.2 Audio Operations in AIVS

# From tts/engine.py — pitch shifting
def _shift_pitch(samples: np.ndarray, pitch: float) -> np.ndarray:
    """Shift pitch without changing duration (resample)."""
    if pitch == 1.0:
        return samples
    # Resample to change pitch
    new_length = int(len(samples) / pitch)
    indices = np.linspace(0, len(samples) - 1, new_length)
    shifted = np.interp(indices, np.arange(len(samples)), samples.astype(np.float64))
    return shifted.astype(np.int16)

# From tts/engine.py — volume adjustment
def _apply_volume(samples: np.ndarray, volume: float) -> np.ndarray:
    """Scale audio samples by volume factor."""
    if volume == 1.0:
        return samples
    scaled = samples.astype(np.float64) * volume
    return np.clip(scaled, -32768, 32767).astype(np.int16)

37.3 WAV File Writing

import wave

def write_wav(path: str, samples: np.ndarray, sample_rate: int = 24000):
    """Write int16 samples to a WAV file."""
    with wave.open(path, "wb") as wf:
        wf.setnchannels(1)       # Mono
        wf.setsampwidth(2)       # 16-bit = 2 bytes
        wf.setframerate(sample_rate)
        wf.writeframes(samples.tobytes())

37.4 Key Takeaways

  • NumPy int16 arrays are the universal audio format in AIVS
  • 24 kHz sample rate is the standard for TTS output
  • Base64 encoding is used to pass audio between subprocesses
  • np.clip() prevents audio clipping (values outside -32768..32767)
  • np.interp() is used for pitch shifting (resampling)
