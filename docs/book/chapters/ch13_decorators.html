13.1 Function Basics — Functions Are Objects

# Pass as argument
def apply(func, value):
    return func(value)

# Return from another function
def make_greeting(greeting):

    return inner

13.2 Writing Your First Decorator

# A decorator is a function that takes a function and returns a new function
import functools

def timer(func):
    """Decorator that prints how long a function takes."""
    @functools.wraps(func)  # Preserves the original function's name/docstring
    def wrapper(*args, **kwargs):
        import time
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"{func.__name__} took {elapsed:.3f}s")
        return result
    return wrapper

# Apply with @ syntax
@timer
def load_model(path):
    import time; time.sleep(0.1)  # Simulate loading
    return "model"

result = load_model("model.onnx")
# Output: load_model took 0.102s

# Equivalent without @ syntax
def load_model2(path):
    return "model"
load_model2 = timer(load_model2)

13.3 Decorators with Arguments

def retry(max_attempts=3, delay=1.0):
    """Decorator that retries a function on failure."""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as exc:
                    if attempt == max_attempts - 1:
                        raise
                    print(f"Attempt {attempt+1} failed, retrying...")
                    time.sleep(delay)
        return wrapper
    return decorator

@retry(max_attempts=5, delay=2.0)
def download_model(url):
    import requests
    return requests.get(url, timeout=30)

13.4 Common Built-in Decorators

class MyClass:
    # @property — computed attribute
    @property
    def display_name(self):
        return f"{self._name} v{self._version}"

    # @classmethod — receives class instead of instance
    @classmethod
    def from_dict(cls, data: dict):
        return cls(data["name"], data["version"])

    # @staticmethod — no access to self or cls
    @staticmethod
    def is_valid(name: str) -> bool:
        return bool(name.strip())

# functools.lru_cache — memoization
@functools.lru_cache(maxsize=128)
def expensive_lookup(voice_id: str) -> dict:
    """Cache expensive database queries."""
    return db.query(voice_id)  # Only runs once per voice_id

# functools.cache (Python 3.9+) — unlimited cache
@functools.cache
def parse_version(version_str: str) -> tuple:
    return tuple(int(x) for x in version_str.split("."))

13.5 Decorator Patterns in AIVS

# The @property decorator pattern used throughout AIVS:
class Settings:
    @property
    def theme(self) -> str:
        return self._data.get("theme", "system")

    @property
    def compute(self) -> str:
        return self._data.get("compute", "auto")

# Context manager decorator pattern:
from contextlib import contextmanager

@contextmanager
def busy_cursor():
    """Show busy cursor during an operation."""
    wx.BeginBusyCursor()
    try:
        yield
    finally:
        wx.EndBusyCursor()

13.6 Key Takeaways

  • A decorator is a function that wraps another function, modifying its behavior
  • Always use @functools.wraps(func) to preserve the original function's metadata
  • Decorators with arguments need three levels of nesting
  • Built-in decorators: @property, @classmethod, @staticmethod, @lru_cache
  • Decorators are used in AIVS for caching, timing, and GUI cursor management
