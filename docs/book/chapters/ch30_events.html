30.1 Basic Event Binding

# Bind a button click
self.start_btn.Bind(wx.EVT_BUTTON, self._on_start)

# Bind with lambda
self.close_btn.Bind(wx.EVT_BUTTON, lambda _: self.Close())

# Bind combo box selection
self.tts_combo.Bind(wx.EVT_COMBOBOX, self._on_tts_changed)

# Bind window close
self.Bind(wx.EVT_CLOSE, self._on_close)

# Bind slider change
self.slider.Bind(wx.EVT_SLIDER, self._on_slider_change)

30.2 Custom Events (Worker → GUI)

# From gui/events.py
# Step 1: Create a unique event type
_SYNTH_STATUS = wx.NewEventType()
EVT_SYNTH_STATUS = wx.PyEventBinder(_SYNTH_STATUS, 1)

# Step 2: Create an event class
class SynthStatusEvent(wx.PyCommandEvent):
    def __init__(self, message: str):
        super().__init__(_SYNTH_STATUS)
        self.message = message

# Step 3: Post from worker thread
def _on_status(self, msg):
    wx.PostEvent(self, SynthStatusEvent(msg))

# Step 4: Handle in GUI (main thread)
self.Bind(EVT_SYNTH_STATUS, self._handle_status)

def _handle_status(self, evt: SynthStatusEvent):
    self.status_label.SetLabel(evt.message)

30.3 Event Propagation

# Events propagate up the widget hierarchy
# Use evt.Skip() to let parent handle the event too

def _on_close(self, evt):
    if self._worker and self._worker.is_alive():
        answer = wx.MessageBox(
            "Recording is running. Stop and close?",
            "Close",
            style=wx.YES_NO | wx.ICON_QUESTION,
        )
        if answer != wx.YES:
            return  # Don't close — consume the event
        self._worker.cancel()
    self._save_tts_to_project()
    self.EndModal(wx.ID_CLOSE)

30.4 Key Takeaways

  • Bind() connects events to handler methods
  • wx.PostEvent() is the only safe way to send events from worker threads
  • Custom events use NewEventType() + PyEventBinder()
  • AIVS has 6 custom event types for synthesis and download communication
  • evt.Skip() allows event propagation to parent widgets
