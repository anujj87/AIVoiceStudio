File System Tests

# Use tempfile for isolated file system tests
class ResolveVoiceFilesTest(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        # Mock paths to use temp directory
        self.patcher = mock.patch(
            "ai_voice_studio.paths.user_data_dir",
            return_value=self.tmp.name,
        )
        self.patcher.start()

    def test_matcha_finds_model(self):
        vdir = self._voice_dir("matcha_en/default")
        open(os.path.join(vdir, "model-steps-3.onnx"), "w").close()
        files = resolve_voice_files({"dir": vdir, "engine": "matcha"})
        assert files["model"].endswith("model-steps-3.onnx")

61.4 Key Takeaways

  • AIVS has 99+ tests across 12 test files
  • Data validation tests ensure catalog integrity
  • Mocking isolates tests from external dependencies (GPU, network)
  • tempfile provides isolated file system for tests
  • The GUI smoke tests verify wx objects can be created without crashes
