10.4 Types of Methods

class AudioFile:
    # Class attribute
    MAX_SIZE_MB = 100

    def __init__(self, path: str, data: bytes):
        self.path = path
        self.data = data

    # Instance method — accesses 'self'
    def size_mb(self) -> float:
        return len(self.data) / (1024 * 1024)

    # Class method — accesses 'cls' (the class itself)
    @classmethod
    def from_file(cls, path: str) -> "AudioFile":
        """Alternative constructor from a file path."""
        with open(path, "rb") as f:
            data = f.read()
        return cls(path, data)  # cls() = AudioFile()

    # Static method — no access to self or cls
    @staticmethod
    def is_valid_format(path: str) -> bool:
        """Check if file extension is supported."""
        return path.lower().endswith((".wav", ".mp3", ".flac"))

# Using all three types
f1 = AudioFile("output.wav", audio_bytes)    # Regular constructor
f2 = AudioFile.from_file("output.wav")       # Class method constructor
print(f1.size_mb())                            # Instance method
print(AudioFile.is_valid_format("test.wav"))  # Static method

class Segment:
    def __init__(self, index: int, title: str, text: str):
        self.index = index
        self.title = title
        self.text = text

    def __repr__(self) -> str:
        """Developer string: Segment(1, 'Chapter 1')"""
        return f"Segment({self.index}, {self.title!r})"

    def __str__(self) -> str:
        """User-facing string: "Chapter 1" """
        return self.title

    def __eq__(self, other) -> bool:
        """Equality comparison."""
        if not isinstance(other, Segment):
            return NotImplemented
        return self.index == other.index

    def __lt__(self, other) -> bool:
        """Less-than comparison (enables sorting)."""
        return self.index < other.index

    def __len__(self) -> int:
        """len(segment) returns text length."""
        return len(self.text)

    def __bool__(self) -> bool:
        """bool(segment) is True if text is non-empty."""
        return bool(self.text.strip())

10.8 Composition vs. Inheritance

# Composition (preferred): objects contain other objects
class RecordingDialog:
    def __init__(self, parent, project_dir, settings, store):
        self.settings = settings    # Has-a Settings
        self.store = store          # Has-a ModelStore
        self._worker = None        # Will have-a SynthesisWorker

# Inheritance (less common): is-a relationship
class DownloadPanel(wx.Panel):    # DownloadPanel IS-A wx.Panel
    def __init__(self, parent, store, downloader):
        super().__init__(parent)  # Call parent's __init__
        self.store = store

Prefer composition over inheritance. AIVS uses inheritance mainly for wxPython widgets (because wx requires it), and composition for everything else. Composition is more flexible and easier to test.

10.9 Key Takeaways

  • Classes define blueprints; objects are instances of classes
  • __init__ initializes new objects — it is NOT the constructor (the constructor is __new__)
  • self refers to the current instance — always the first parameter
  • Instance attributes are per-object; class attributes are shared
  • @property provides computed/validated attribute access
  • Dunder methods (__repr__, __eq__, __len__) make objects integrate with Python's built-in functions
  • Prefer composition — objects contain other objects rather than inheriting from them
  • The AIVS project has classes for every major subsystem: Settings, ModelStore, CloneClient, RecordingDialog, etc.
