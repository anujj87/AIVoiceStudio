70.1 Lazy Loading

# Heavy imports are deferred until needed
def heavy_function():
    import numpy  # Only imported when function is called
    return np.array([1, 2, 3])

# TTS engines are loaded on first use
_engine_cache: dict[str, TtsEngine] = {}

def get_engine(voice_entry):
    key = voice_entry["voice"]
    if key not in _engine_cache:
        _engine_cache[key] = TtsEngine(voice_entry)  # Expensive!
    return _engine_cache[key]

70.3 Audio Memory Management

# Audio arrays can be large (1 sec = 48KB at 24kHz/16-bit)
# A 10-minute recording = ~28 MB

# Use appropriate dtypes

# Release memory promptly
del large_array
gc.collect()

# Base64 encoding doubles size temporarily
wav_b64 = base64.b64encode(samples.tobytes())  # 2x memory
# Decode immediately after receiving
samples = np.frombuffer(base64.b64decode(wav_b64), dtype=np.int16)

70.4 Thread Optimization

# Daemon threads die with the main process (no zombie processes)
t = threading.Thread(target=work, daemon=True)

# Lock granularity — don't hold locks longer than needed
with self._lock:
    # Minimal critical section
    self._proc.stdin.write(data)

# Use time.sleep(0.1) in poll loops to avoid CPU spinning
while self._pause_event.is_set():
    time.sleep(0.1)  # Don't burn CPU while paused

70.5 Key Takeaways

  • Lazy loading defers expensive imports until needed
  • Engine caching prevents reloading 50-500 MB models
  • int16 dtype is the standard for audio (2 bytes per sample)
  • Daemon threads prevent zombie processes
  • Lock granularity — hold locks for minimal time
