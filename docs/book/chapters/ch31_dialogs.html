31.1 Modal Dialogs

# wx.Dialog — blocks the parent until closed
class MyDialog(wx.Dialog):
    def __init__(self, parent):
        super().__init__(parent, title="Settings", size=(600, 400))
        panel = wx.Panel(self)
        sizer = wx.BoxSizer(wx.VERTICAL)
        sizer.Add(wx.StaticText(panel, label="Configure settings here"))
        ok_btn = wx.Button(panel, wx.ID_OK, label="Save")
        cancel_btn = wx.Button(panel, wx.ID_CANCEL, label="Cancel")
        btn_sizer = wx.BoxSizer(wx.HORIZONTAL)
        btn_sizer.Add(ok_btn, 0, wx.ALL, 5)
        btn_sizer.Add(cancel_btn, 0, wx.ALL, 5)
        sizer.Add(btn_sizer)
        panel.SetSizer(sizer)

# Show modal dialog
dlg = MyDialog(self)
result = dlg.ShowModal()  # Blocks until closed
if result == wx.ID_OK:
    # User clicked Save
    pass
dlg.Destroy()  # Always destroy after use

31.3 The New Project Wizard

# From gui/new_project_wizard.py
from wx.adv import Wizard, WizardPage

class NewProjectWizard(Wizard):
    def __init__(self, parent, settings, store):
        super().__init__(parent, title="New Project - AI Voice Studio")

        # Page 1: project details
        self.page_details = _DetailsPage(self)
        self.page_details.name_ctrl = wx.TextCtrl(...)
        self.page_details.open_btn = wx.Button(...)

        # Page 2: audio mode selection
        self.page_mode = _ModePage(self, settings)
        self.page_mode.radios = [...]
        self.page_mode.description = wx.StaticText(...)

        self.GetPageAreaSizer().Add(self.page_details)

        self.Bind(EVT_WIZARD_FINISHED, self._on_finish)

    def _on_finish(self, _):
        """Called when user clicks Finish."""
        name = self.page_details.name_ctrl.GetValue().strip()
        mode = self.page_mode.selected()

        # Parse document and create project
        doc = parse_document(self.source_path)
        segments = split_document(doc, mode)
        project.create_project(pdir, name, self.source_path, mode, segments)
        self.EndModal(wx.ID_OK)

        # Open the recording window
        dlg = RecordingDialog(self.GetParent(), pdir, self.settings, self.store)
        dlg.ShowModal()

31.4 Key Takeaways

  • Modal dialogs block the parent — use for settings, wizards, confirmations
  • ShowModal() returns wx.ID_OK or wx.ID_CANCEL
  • Always call Destroy() on dialogs (or use with statement)
  • wx.adv.Wizard provides multi-step wizard pages with Next/Back/Finish
  • The AIVS wizard has 2 pages: Details (name, document, punctuation) and Mode (splitting method)
