Chapter 44

Paths, Settings & Constants

Current implementation (2026.3.1). paths.py now also lays out tts_envs\<engine> (one Python environment per pip-installed TTS engine) alongside addon_env, runtime, ffmpeg and the model and project folders. settings.py gained: preview_compute (the Compute choice of each Preview button, keyed by category), clone_engines (engine, device, options.<engine>), omnivoice_server (host, port, device, steps, concurrency, API key, CORS), daisy, recording.per_tts, and terms_version — the last one written only when the user accepts the first-launch terms, and compared with constants.TERMS_VERSION on every start. constants.py carries APP_VERSION and TERMS_VERSION, which the installer version must match.

Learning Objectives

  • Understand the AIVS path resolution system
  • Master the Settings class with dot-notation access
  • See how constants organize application configuration

44.1 Path Resolution

# From paths.py
def user_data_dir() -> str:
    """Return %APPDATA%/AIVoiceStudio on Windows."""
    if sys.platform == "win32":
        base = os.environ.get("APPDATA", os.path.expanduser("~"))
    else:
        base = os.path.join(os.path.expanduser("~"), ".config")
    d = os.path.join(base, "AIVoiceStudio")
    os.makedirs(d, exist_ok=True)
    return d

def models_dir() -> str:
    """Return the directory where TTS models are stored."""
    d = os.path.join(user_data_dir(), "models")
    os.makedirs(d, exist_ok=True)
    return d

def project_dir(name: str) -> str:
    """Return the directory for a project."""
    d = os.path.join(user_data_dir(), "projects", sanitize_filename(name))
    os.makedirs(d, exist_ok=True)
    return d

def settings_file() -> str:
    return os.path.join(user_data_dir(), "settings.json")

def log_file() -> str:
    return os.path.join(user_data_dir(), "app.log")

44.2 Settings with Dot-Notation

# From settings.py
class Settings:
    _defaults = {
        "theme": "system",
        "audio_mode": "page_with_h1",
        "recording.rate": 1.0,
        "recording.pitch": 1.0,
        "recording.volume": 1.0,
        "recording.output_format": "wav",
        "recording.punctuation": "default",
        "compute": "auto",
        "developer_mode": False,
        "recent_projects": [],
    }

    def __init__(self, path=None):
        self._path = path or settings_file()
        self._data = dict(self._defaults)
        self._load()

    def get(self, key, default=None):
        # Simple: just use the key as-is
        return self._data.get(key, default)

    def set(self, key, value):
        self._data[key] = value
        self.save()

    def save(self):
        os.makedirs(os.path.dirname(self._path), exist_ok=True)
        with open(self._path, "w", encoding="utf-8") as f:
            json.dump(self._data, f, indent=2, ensure_ascii=False)

    @property
    def theme(self) -> str:
        return self._data.get("theme", "system")

    @property
    def compute(self) -> str:
        return self._data.get("compute", "auto")

44.4 Key Takeaways

  • paths.py centralizes all file locations (user data, models, projects, logs)
  • settings.py provides JSON persistence with property-based access
  • constants.py eliminates magic numbers and strings across the codebase
  • Settings are loaded at startup and saved immediately on change
