20.2 The subprocess Module

import subprocess

# Run a command and wait
result = subprocess.run(
    ["python", "-c", "print('hello')"],
    capture_output=True,
    text=True,
)
print(result.stdout)       # "hello\n"
print(result.returncode)   # 0

# Start a long-running process
proc = subprocess.Popen(
    ["python", "-m", "worker"],
    stdin=subprocess.PIPE,
    stdout=subprocess.PIPE,
    text=True,
)

# Communicate via stdin/stdout
proc.stdin.write('{"cmd": "ping"}\n')
proc.stdin.flush()
response = proc.stdout.readline()  # Read one line

# Terminate
proc.terminate()
proc.wait(timeout=5)

20.3 AIVS Subprocess Architecture

AI Voice Studio uses subprocesses to isolate heavy runtimes (PyTorch, ONNX) from the main GUI process. This is a critical architectural decision: 

# From clone/__init__.py — CloneClient manages a subprocess
class CloneClient:
    def _ensure_proc(self):
        if self._proc is not None and self._proc.poll() is None:
            return
        cmd = [sys.executable, "-m", "ai_voice_studio.clone.worker"]
        self._proc = subprocess.Popen(
            cmd,
            stdin=subprocess.PIPE,
            stdout=subprocess.PIPE,
            stderr=subprocess.DEVNULL,
            text=True,
            bufsize=1,
        )
        # Verify the worker started
        resp = self._request({"cmd": "ping"})
        if not resp.get("ok"):
            raise CloneEngineError("Worker failed to start")

    def _request(self, req: dict) -> dict:
        """Send JSON request, read JSON response."""
        with self._lock:
            req["id"] = self._next_id
            self._next_id += 1
            self._proc.stdin.write(json.dumps(req) + "\n")
            self._proc.stdin.flush()
            line = self._proc.stdout.readline()
            return json.loads(line)

# Protocol: JSON lines over stdin/stdout
# Request:  {"cmd": "synthesize", "text": "Hello", "speed": 1.0}
# Response: {"ok": true, "wav": "base64..."}

    creationflags = getattr(subprocess, "CREATE_NO_WINDOW", 0)

    cmd,

20.5 Key Takeaways

  • Use subprocess when you need true isolation (different runtime, memory space)
  • AIVS uses subprocesses for TTS engines to prevent runtime conflicts
  • JSON lines protocol over stdin/stdout is the communication pattern
  • Always use threading.RLock to synchronize access to the subprocess
