Chapter 51

Main Frame & Menu System

Current implementation (2026.3.1). The menus now read: File (New Project, Open Project), Edit (Resume Recording, Restart Project..., Restart All Recording, Start Selected Recording..., Remove Project...), Tools (Record, Settings) and Help (Read Me, User Guide, Addon Development Guide, Accessibility Guidelines, Python & wxPython Book F1, Third-Party Licences, About). The picker behind Start Selected Recording... is implemented in this module (_picker_entries + _StartRecordingDialog) and is described in Chapter 80. Every document opened from Help goes through one _open_doc helper, so source and frozen builds behave the same.

Learning Objectives

  • Understand the MainFrame class and its responsibilities
  • See the menu system, keyboard shortcuts, and recent projects
  • Understand the application lifecycle

class MainFrame(wx.Frame):
    def __init__(self, parent, settings: Settings, store: ModelStore):
        super().__init__(parent, title=f"{APP_NAME} v{APP_VERSION}", size=(800, 600))
        self.settings = settings
        self.store = store
        self._build_menu()
        self._build_ui()
        self._bind_events()
        self.Centre()
        self.SetStatusText("Ready")

    def _build_menu(self):
        # File menu with New, Open, Recent, Exit
        # Edit menu (future)
        # Tools menu with Record
        # Settings menu with Settings dialog
        # Help menu with About

    def _build_ui(self):
        # Welcome panel with project info
        panel = wx.Panel(self)
        sizer = wx.BoxSizer(wx.VERTICAL)
        welcome = wx.StaticText(panel, label="Welcome to AI Voice Studio")
        sizer.Add(welcome, 0, wx.ALL | wx.ALIGN_CENTER, 20)
        panel.SetSizer(sizer)

    def _bind_events(self):
        # Menu event handlers
        self.Bind(wx.EVT_MENU, self._on_new_project, id=ID_NEW_PROJECT)
        self.Bind(wx.EVT_MENU, self._on_open_project, id=ID_OPEN_PROJECT)
        self.Bind(wx.EVT_MENU, self._on_settings, id=ID_SETTINGS)
        self.Bind(wx.EVT_MENU, self._on_record, id=ID_RECORD)
        self.Bind(wx.EVT_MENU, self._on_exit, id=wx.ID_EXIT)
        self.Bind(wx.EVT_MENU, self._on_about, id=wx.ID_ABOUT)

    def _on_new_project(self, _):
        dlg = NewProjectWizard(self, self.settings, self.store)
        dlg.ShowModal()
        dlg.Destroy()

    def _on_settings(self, _):
        dlg = SettingsDialog(self, self.settings, self.store)
        dlg.ShowModal()
        dlg.Destroy()
        apply_theme(self, self.settings.theme)

51.3 Application Lifecycle

# From main.py
def main() -> int:
    settings = Settings()
    developer_mode = settings.get("developer_mode", False)
    _setup_logging(developer_mode)
    _init_python_runtime()
    _init_addons()

    app = wx.App(False)
    store = ModelStore()
    frame = MainFrame(None, settings, store)
    apply_theme(frame, settings.theme)
    frame.Show()
    result = app.MainLoop()
    return result or 0

51.4 Key Takeaways

  • MainFrame is the application hub — menus, status bar, event routing
  • Five menus: File, Edit, Tools, Settings, Help
  • Keyboard shortcuts follow Windows conventions (Ctrl+key)
  • The application starts: Settings → Logging → Runtime → Addons → wx.App → MainFrame
  • apply_theme() is called after settings changes to update the UI
