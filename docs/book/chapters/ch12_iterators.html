Iterators & Generators

12.1 The Iterator Protocol

# Every iterable has __iter__ that returns an iterator
# Every iterator has __next__ that returns the next value

class VoiceIterator:
    """Iterates over voice entries in the catalog."""
    def __init__(self, catalog):
        self._catalog = catalog
        self._tts_idx = 0
        self._lang_idx = 0
        self._var_idx = 0
        self._voice_idx = 0

    def __iter__(self):
        return self  # Iterators return themselves

    def __next__(self):
        # Navigate through nested structure
        while self._tts_idx < len(self._catalog):
            tts = self._catalog[self._tts_idx]
            langs = tts.get("languages", [])
            while self._lang_idx < len(langs):
                lang = langs[self._lang_idx]
                variants = lang.get("variants", [])
                while self._var_idx < len(variants):
                    variant = variants[self._var_idx]
                    voices = variant.get("voices", [])
                    while self._voice_idx < len(voices):
                        voice = voices[self._voice_idx]
                        self._voice_idx += 1
                        return {"tts": tts["id"], "voice": voice}
                    self._var_idx += 1
                    self._voice_idx = 0
                self._lang_idx += 1
                self._var_idx = 0
            self._tts_idx += 1
            self._lang_idx = 0
        raise StopIteration

# Usage
for voice in VoiceIterator(catalog):
    print(voice)

12.2 Generator Functions

Generators use yield to produce values lazily, one at a time:

def voice_entries(catalog):
    """Yield all voice entries from the catalog (lazy)."""
    for tts in catalog:
        for lang in tts.get("languages", []):
            for variant in lang.get("variants", []):
                for voice in variant.get("voices", []):
                    yield {
                        "tts": tts["id"],
                        "language": lang["code"],
                        "variant": variant["id"],
                        "voice": voice,
                    }

# Usage — memory efficient (no list in memory)
for entry in voice_entries(catalog):
    print(entry["voice"]["id"])

# Can convert to list if needed
all_entries = list(voice_entries(catalog))

yield from — Delegating to Sub-generators

def all_paths(directory):
    """Recursively yield all file paths."""
    for entry in os.scandir(directory):
        if entry.is_file():
            yield entry.path
        elif entry.is_dir():
            yield from all_paths(entry.path)  # Delegate to sub-generator

12.3 Generator Expressions

# List comprehension creates list in memory
big_list = [x ** 2 for x in range(1_000_000)]  # ~8MB in memory

# Generator expression — lazy evaluation (almost no memory)
big_gen = (x ** 2 for x in range(1_000_000))  # One number at a time

# Common pattern: sum/count/min/max on generators
total_chars = sum(len(v["text"]) for v in segments)
max_length = max(len(s["text"]) for s in segments)
has_empty = any(not s["text"].strip() for s in segments)

12.4 itertools — Advanced Iteration

import itertools

# chain — concatenate iterables
all_voices = list(itertools.chain(piper_voices, kokoro_voices))

# islice — slice an iterator
first_10 = list(itertools.islice(voice_iterator, 10))

# groupby — group consecutive elements
from itertools import groupby
voices_sorted = sorted(voices, key=lambda v: v["tts"])
for tts_id, group in groupby(voices_sorted, key=lambda v: v["tts"]):
    print(f"{tts_id}: {len(list(group))} voices")

# product — Cartesian product
engines = ["piper", "kokoro"]
languages = ["en", "zh"]
for engine, lang in itertools.product(engines, languages):
    print(f"{engine}/{lang}")

# cycle — infinite cycling
spinner = itertools.cycle(["|", "/", "-", "\\"])
for _ in range(8):
    print(f"\r{next(spinner)} Loading...", end="")

12.5 Real Example: Segmented File Reading

# From AIVS: reading text files in chunks for large documents
def read_in_chunks(file_path: str, chunk_size: int = 4096):
    """Read a file in chunks — memory efficient."""
    with open(file_path, "r", encoding="utf-8") as f:
        while True:
            chunk = f.read(chunk_size)
            if not chunk:
                break
            yield chunk

# Count words without loading entire file
def count_words(file_path: str) -> int:
    return sum(chunk.count(" ") + 1 for chunk in read_in_chunks(file_path))

12.6 Key Takeaways

  • Generators are the Pythonic way to create iterators — much simpler than implementing __iter__/__next__
  • yield pauses the function and produces a value; yield from delegates to sub-generators
  • Generators are lazy — they produce values on demand, saving memory
  • Generator expressions are like list comprehensions but with parentheses: (x for x in ...)
  • itertools provides powerful iteration building blocks (chain, groupby, product, etc.)
