63.2 End-to-End Test Patterns

# Test: parse document → split → create project → save/load
def test_full_project_lifecycle(self):
    # 1. Create a temp text file
    txt = os.path.join(self.tmp.name, "test.txt")
    with open(txt, "w") as f:
        f.write("Chapter 1\n\nHello world.\n\nChapter 2\n\nGoodbye.")

    # 2. Parse
    doc = parse_document(txt)
    assert len(doc.blocks) >= 2

    # 3. Split
    segments = split_document(doc, MODE_PAGE_WITH_H1)
    assert len(segments) >= 2

    # 4. Create project
    pdir = os.path.join(self.tmp.name, "project")
    project.create_project(pdir, "Test", txt, "page_with_h1",
        [{"index": s.index, "title": s.title, "text": s.text}
         for s in segments])

    # 5. Load and verify
    data = project.load_project(pdir)
    assert data["name"] == "Test"
    assert len(data["segments"]) >= 2

    # 6. Mark segment done
    project.mark_segment_done(pdir, 1, "test.wav")
    data = project.load_project(pdir)
    assert data["segments"][0]["status"] == "done"

63.3 Key Takeaways

  • Smoke tests verify objects can be created without crashes
  • End-to-end tests exercise the full pipeline (parse → split → save → load)
  • AIVS tests avoid actual TTS synthesis (too slow, requires models)
  • The test suite runs in under 30 seconds
