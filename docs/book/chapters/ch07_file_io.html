7.1 Opening and Closing Files

# The "classic" way (don't do this in modern Python)
f = open("data.txt", "r")
content = f.read()
f.close()  # Must close manually!

# The correct way — context manager (auto-closes)
with open("data.txt", "r") as f:
    content = f.read()
# f is automatically closed here, even if an exception occurs

File Modes

ModeDescription
"r"Read (default). File must exist.
"w"Write. Creates or truncates file.
"a"Append. Creates if doesn't exist.
"x"Exclusive create. Fails if file exists.
"r+"Read + write. File must exist.
"rb"Read binary.
"wb"Write binary.

AIVS Settings Pattern

# From ai_voice_studio/settings.py
def load(self) -> None:
    """Load settings from disk, merging with defaults."""
    if os.path.isfile(self._path):
        with open(self._path, "r", encoding="utf-8") as f:
            stored = json.load(f)
        # Merge: defaults + stored values
        self._data = {**self._defaults, **stored}

def save(self) -> None:
    """Save current settings to disk."""
    os.makedirs(os.path.dirname(self._path), exist_ok=True)
    with open(self._path, "w", encoding="utf-8") as f:
        json.dump(self._data, f, indent=2, ensure_ascii=False)

7.8 File I/O in AIVS

# From project.py — atomic JSON write
def save_project(project_dir: str, data: dict) -> None:
    """Save project.json atomically (write to temp, then rename)."""
    import tempfile
    path = os.path.join(project_dir, "project.json")
    # Write to a temp file first (atomic on same filesystem)
    fd, tmp = tempfile.mkstemp(dir=project_dir, suffix=".tmp")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2, ensure_ascii=False)
        # Atomic rename (on same filesystem)
        os.replace(tmp, path)
    except:
        os.unlink(tmp)
        raise

# From tts/downloader.py — streaming download with progress
def _download_file(url: str, dest: str, progress=None) -> None:
    """Download a file with progress reporting."""
    import requests
    with requests.get(url, stream=True) as resp:
        total = int(resp.headers.get("content-length", 0))
        downloaded = 0
        with open(dest, "wb") as f:
            for chunk in resp.iter_content(chunk_size=8192):
                f.write(chunk)
                downloaded += len(chunk)
                if progress:
                    progress(downloaded, total)

7.10 Key Takeaways

  • Always use with statements for file I/O (context managers auto-close)
  • Always specify encoding="utf-8" for text files
  • JSON is the standard format for settings and project data in AIVS
  • Atomic writes (write to temp file, then rename) prevent corruption
  • pathlib is the modern way to handle file paths
  • For streaming large files, read in chunks to avoid loading everything into memory
