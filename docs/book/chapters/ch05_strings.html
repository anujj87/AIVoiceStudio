5.6 Strings in TTS Processing (AIVS)

The TTS engine requires careful text preparation before synthesis:

# From documents/splitter.py — text segmentation for TTS
def _split_sentences(text: str) -> list[str]:
    """Split text into sentences for TTS processing."""
    import re
    # Split on sentence-ending punctuation
    sentences = re.split(r'(?<=[.!?])\s+', text)
    return [s.strip() for s in sentences if s.strip()]

# From util.py — filename sanitization
def sanitize_filename(name: str, max_length: int = 80) -> str:
    """Remove unsafe characters from filenames."""
    import re
    name = re.sub(r'[<>:"/\\|?*]', '_', name)
    name = re.sub(r'\s+', ' ', name).strip()
    if len(name) > max_length:
        name = name[:max_length].rstrip()
    return name

# From documents/parsers.py — text normalization
def _normalize_text(text: str) -> str:
    """Normalize text for TTS processing."""
    # Remove excessive whitespace
    text = re.sub(r'\n{3,}', '\n\n', text)
    # Normalize Unicode
    text = text.strip()
    return text

# From tts/catalog.py — building voice labels
def _build_voice_label(voice: dict) -> str:
    """Build a display label like 'af_bella (af_bella)'."""
    name = voice.get("name", voice["id"])
    voice_id = voice["id"]
    if name != voice_id:
        return f"{name} ({voice_id})"
    return name

5.9 Key Takeaways

  • Strings are immutable — every method returns a new string
  • f-strings are the preferred formatting method (concise, readable, fast)
  • str methods cover virtually all text processing needs
  • str.split() and str.join() are your best friends for text manipulation
  • Unicode is built-in — Python 3 handles international text natively
  • The AIVS project uses string processing for text normalization, filename sanitization, and voice label formatting
