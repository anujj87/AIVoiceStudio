14.1 The Context Manager Protocol

# Context managers implement __enter__ and __exit__
class ManagedEngine:
    """TTS engine that cleans up when done."""

    def __init__(self, engine_id: str):
        self.engine_id = engine_id
        self._model = None

    def __enter__(self):
        """Called when entering the 'with' block."""
        print(f"Loading {self.engine_id}...")
        self._model = "loaded"
        return self  # This is assigned to 'as' variable

    def __exit__(self, exc_type, exc_val, exc_tb):
        """Called when exiting the 'with' block (always)."""
        print(f"Unloading {self.engine_id}...")
        self._model = None
        return False  # Don't suppress exceptions

# Usage
with ManagedEngine("piper") as engine:
    print(f"Using {engine.engine_id}")
    # ... use the engine ...
# "Unloading piper..." is printed here, even if an exception occurs

14.2 @contextmanager Decorator

from contextlib import contextmanager

@contextmanager
def managed_engine(engine_id: str):
    """Simpler context manager using @contextmanager."""
    print(f"Loading {engine_id}...")
    engine = "loaded"
    try:
        yield engine  # This is the 'as' value
    finally:
        print(f"Unloading {engine_id}...")

# Usage (identical to the class version)
with managed_engine("piper") as engine:
    print(f"Using {engine}")

14.3 Context Managers in AIVS

# GUI busy cursor management
from contextlib import contextmanager

@contextmanager
def busy_cursor():
    """Show busy cursor during document parsing."""
    wx.BeginBusyCursor()
    try:
        yield
    finally:
        wx.EndBusyCursor()

# Usage in NewProjectWizard
with busy_cursor():
    doc = parse_document(self.source_path)
    segments = split_document(doc, mode)

# Thread-safe lock management
class CloneClient:
    def _request(self, req: dict) -> dict:
        with self._lock:  # Thread-safe access
            # ... send request and read response

# Temporary directory for downloads
import tempfile
with tempfile.TemporaryDirectory() as tmpdir:
    # Download to tmpdir, process, then it's auto-cleaned
    download_to(tmpdir)

# contextlib.suppress — ignore specific exceptions
from contextlib import contextlib
with contextlib.suppress(FileNotFoundError):
    os.unlink(temp_file)  # Ignore if file doesn't exist

14.4 Key Takeaways

  • Context managers ensure cleanup always happens, even on exceptions
  • @contextmanager is simpler than implementing __enter__/__exit__
  • Use context managers for files, locks, subprocesses, GUI state
  • The AIVS project uses context managers for thread locks, temp directories, and GUI busy cursor
