38.1 HTTP Basics with requests

import requests

# GET request
resp = requests.get("https://api.example.com/voices")
print(resp.status_code)  # 200
print(resp.json())       # Parsed JSON

# POST request
resp = requests.post(url, json={"text": "Hello"})

# Download with streaming
with requests.get(url, stream=True) as resp:
    total = int(resp.headers.get("content-length", 0))
    downloaded = 0
    with open(dest, "wb") as f:
        for chunk in resp.iter_content(chunk_size=8192):
            f.write(chunk)
            downloaded += len(chunk)

# Error handling
try:
    resp = requests.get(url, timeout=30)
    resp.raise_for_status()  # Raises HTTPError for 4xx/5xx
except requests.ConnectionError:
    print("Connection failed")
except requests.Timeout:
    print("Request timed out")

38.2 AIVS Model Downloader

# From tts/downloader.py
def _download_file(url, dest, progress=None, cancel_event=None):
    """Download a file with progress and cancellation."""
    with requests.get(url, stream=True, timeout=60) as resp:
        resp.raise_for_status()
        total = int(resp.headers.get("content-length", 0))
        downloaded = 0
        with open(dest, "wb") as f:
            for chunk in resp.iter_content(chunk_size=65536):
                if cancel_event and cancel_event.is_set():
                    raise DownloadCancelled()
                f.write(chunk)
                downloaded += len(chunk)
                if progress:
                    progress("downloading", downloaded, total)

38.3 Key Takeaways

  • requests.get(url, stream=True) for large file downloads
  • iter_content(chunk_size) for memory-efficient streaming
  • raise_for_status() converts HTTP errors to exceptions
  • AIVS downloads TTS voice models (50-500 MB) with progress callbacks
  • Cancellation is checked between each chunk via threading.Event
