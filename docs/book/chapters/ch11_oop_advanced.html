class SherpaEngine(BaseEngine):
    """sherpa-onnx based TTS engine."""
    def __init__(self, voice_entry: dict, model_files: dict):
        super().__init__(voice_entry)  # Call parent's __init__
        self.model_files = model_files
        self._model = None

    def synthesize(self, text, speed=1.0):
        if self._model is None:
            self._model = self._load_model()
        return self._model.generate(text, speed=speed)

class ClonedVoiceEngine(BaseEngine):
    """XTTS v2 voice cloning engine."""
    def __init__(self, voice_entry: dict, client: "CloneClient"):
        super().__init__(voice_entry)
        self._client = client
        self.sample_wav = voice_entry.get("sample")

    def synthesize(self, text, speed=1.0):
        return self._client.synthesize(text, self.sample_wav, speed=speed)

    def close(self):
        self._client.close()  # Override: shut down worker process

11.4 Mixins

# Mixins provide reusable behavior
class LoggingMixin:
    """Adds a logger to any class."""
    @property
    def log(self):
        import logging
        return logging.getLogger(type(self).__name__)

class SingletonMixin:
    """Makes a class a singleton."""
    _instance = None
    _lock = threading.Lock()

    def __new__(cls, *args, **kwargs):
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
        return cls._instance

# Usage
class AddonManager(SingletonMixin, LoggingMixin):
    def discover(self):
        self.log.info("Discovering addons...")

11.7 Key Takeaways

  • Abstract base classes define interfaces that subclasses must implement
  • Dataclasses auto-generate boilerplate methods — use them for data containers
  • Mixins provide reusable behavior without deep inheritance hierarchies
  • super() should always be used — never hardcode parent class names
  • Enums represent fixed sets of values (compute modes, project types)
  • Composition over inheritance remains the golden rule
