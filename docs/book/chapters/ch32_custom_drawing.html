32.1 Device Context (DC) Basics

import wx

class WaveformPanel(wx.Panel):
    """Custom panel that draws an audio waveform."""

    def __init__(self, parent):
        super().__init__(parent)
        self._samples = []
        self.Bind(wx.EVT_PAINT, self._on_paint)

    def set_samples(self, samples):
        self._samples = samples
        self.Refresh()  # Trigger repaint

    def _on_paint(self, event):
        dc = wx.PaintDC(self)
        width, height = self.GetSize()

        # Clear background
        dc.SetBackground(wx.Brush(wx.WHITE))
        dc.Clear()

        if not self._samples:
            return

        # Draw waveform
        dc.SetPen(wx.Pen(wx.BLUE, 1))
        mid = height // 2
        step = max(1, len(self._samples) // width)

        for x in range(width):
            idx = int(x * len(self._samples) / width)
            sample = self._samples[idx] / 32768.0
            y = int(mid - sample * mid * 0.9)
            dc.DrawPoint(x, y)

32.3 Themed Drawing in AIVS

# From gui/theme.py — applying colors to controls
def _paint(window: wx.Window, colors: dict) -> None:
    """Apply theme colors to a control."""
    try:
        if isinstance(window, (wx.Panel, wx.Dialog, wx.Frame)):
            window.SetBackgroundColour(colors["bg"])
        elif isinstance(window, (wx.StaticText, wx.CheckBox)):
            window.SetForegroundColour(colors["fg"])
        elif isinstance(window, (wx.TextCtrl, wx.ComboBox)):
            window.SetBackgroundColour(colors["control_bg"])
            window.SetForegroundColour(colors["control_fg"])
    except Exception:
        pass  # Some controls don't support color changes

32.4 Key Takeaways

  • DC (Device Context) is the drawing API — paint, draw, fill
  • BufferedDC prevents flicker by double-buffering
  • Bind(EVT_PAINT) for custom drawing; call Refresh() to trigger repaint
  • AIVS uses DC drawing for theme application across all widgets
