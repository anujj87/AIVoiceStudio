36.1 wx.adv.Wizard

from wx.adv import Wizard, WizardPage, EVT_WIZARD_FINISHED, EVT_WIZARD_PAGE_CHANGED

class NewProjectWizard(Wizard):
    def __init__(self, parent, settings, store):
        super().__init__(parent, title="New Project")

        # Create pages
        self.page_details = _DetailsPage(self)
        self.page_mode = _ModePage(self, settings)

        # Wire up the page area
        self.GetPageAreaSizer().Add(self.page_details)

        # Events
        self.Bind(EVT_WIZARD_FINISHED, self._on_finish)
        self.Bind(EVT_WIZARD_PAGE_CHANGED, self._on_page_changed)

class _DetailsPage(WizardPage):
    def __init__(self, wizard):
        super().__init__(wizard)
        self.wizard = wizard
        self._build_ui()

    def GetNext(self):
        return self.wizard.page_mode  # Next page

    def GetPrev(self):
        return None  # This is the first page

36.4 Key Takeaways

  • wx.adv.Wizard provides multi-page wizard UI with Next/Back/Finish
  • wx.adv.TaskBarIcon for system tray integration
  • AIVS uses the Wizard for new project creation (2 pages)
  • The Wizard pattern: create pages, link them with GetNext()/GetPrev(), handle EVT_WIZARD_FINISHED
