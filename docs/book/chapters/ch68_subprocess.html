68.2 JSON-Lines IPC Pattern

# All workers follow the same pattern:
# 1. Worker reads JSON from stdin (one line per request)
# 2. Worker writes JSON to stdout (one line per response)
# 3. App manages the subprocess lifecycle

# Worker loop
for line in sys.stdin:
    req = json.loads(line.strip())
    if req["cmd"] == "quit": break
    if req["cmd"] == "ping":
        respond({"ok": True, "engine": True})
    if req["cmd"] == "synthesize":
        audio = synthesize(req["text"])
        respond({"ok": True, "wav": base64.b64encode(audio)})

# App-side request
def _request(self, req):
    with self._lock:
        req["id"] = self._next_id
        self._proc.stdin.write(json.dumps(req) + "\n")
        line = self._proc.stdout.readline()
        return json.loads(line)

68.3 Frozen App Worker Embedding

# In frozen apps, PyInstaller bundles .py files into a PYZ archive
# The managed venv's Python can't read .pyz files
# Solution: embed worker source as a string constant

_WORKER_SOURCE = r'''
import json, sys, numpy as np
def main():
    for line in sys.stdin:
        req = json.loads(line.strip())
        if req["cmd"] == "quit": break
        if req["cmd"] == "synthesize":
            # ... synthesize and respond ...
'''

def _read_worker_source() -> str:
    if getattr(sys, "frozen", False):
        return _WORKER_SOURCE  # Use embedded source
    # Dev mode: read from actual file
    with open("worker.py") as f:
        return f.read()

68.4 Key Takeaways

  • Subprocess isolation prevents runtime conflicts (ONNX vs PyTorch)
  • JSON-lines protocol is the universal IPC pattern
  • Embedded worker source solves the frozen app problem
  • All workers follow the same lifecycle: start → ping → process → quit
